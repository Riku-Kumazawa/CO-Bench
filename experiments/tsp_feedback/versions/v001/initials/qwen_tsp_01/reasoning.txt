def held_karp(dist, n):
    # Fix city 0 as start
    # dp[mask][j] = min cost to start at 0, visit all cities in mask, end at j
    # mask is over cities 1..n-1
    
    m = n - 1  # number of cities excluding start
    INF = float('inf')
    
    # dp is a 2D array: dp[mask][j] where mask is bitmask over m cities, j is 0..m-1
    # j corresponds to city j+1
    dp = np.full((1 << m, m), INF, dtype=np.float64)
    parent = np.full((1 << m, m), -1, dtype=np.int32)
    
    # Base case: mask with only city j visited
    for j in range(m):
        dp[1 << j, j] = dist[0, j + 1]
    
    # Fill DP
    for mask in range(1, 1 << m):
        for j in range(m):
            if not (mask & (1 << j)):
                continue
            if mask == (1 << j):
                continue  # base case
            # Try all possible previous cities i
            best = INF
            best_i = -1
            for i in range(m):
                if i == j or not (mask & (1 << i)):
                    continue
                prev_mask = mask & ~(1 << j)
                val = dp[prev_mask, i] + dist[i + 1, j + 1]
                if val < best:
                    best = val
                    best_i = i
            dp[mask, j] = best
            parent[mask, j] = best_i
    
    # Find best ending city
    full_mask = (1 << m) - 1
    best_cost = INF
    best_j = -1
    for j in range(m):
        cost = dp[full_mask, j] + dist[j + 1, 0]
        if cost < best_cost:
            best_cost = cost
            best_j = j
    
    # Reconstruct tour
    tour = [0]
    mask = full_mask
    j = best_j
    while j != -1:
        tour.append(j + 1)
        prev_j = parent[mask, j]
        mask = mask & ~(1 << j)
        j = prev_j
    
    tour.reverse()
    # tour[0] should be 0
    return tour
